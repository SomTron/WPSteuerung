# Abhaengigkeiten der Heizungssteuerung (WPSteuerung)
# Installiert via: pip install -r requirements.txt
#
# Raspberry-Pi-spezifische Hardware-Pakete (RPi.GPIO, RPLCD, smbus2)
# liegen separat in requirements-pi.txt, da sie ausserhalb des Pi nicht
# installierbar sind und in Tests/Entwicklung gemockt werden (conftest.py).
# Deployment auf dem Pi:  pip install -r requirements-pi.txt
#
# PINNING (Stand 11.11.2026): Die Versionen sind exakt angegeben, damit ein
# Build reproduzierbar ist. Vorher standen die Pakete ohne Version - ein
# `pip install` auf dem Pi konnte damit still eine andere Major-Version
# holen (z. B. pandas 2 -> 3 mit geaendertem API) und die Steuerung erst
# nach dem naechsten Neustart brechen.
#
# Die Pins entsprechen dem Stand, gegen den die Tests laufen (lokal
# verifiziert). Beim Anheben bitte ALLE Pins pruefen - die Untergrenzen der
# Pakete sind unterschiedlich:
#   * matplotlib 3.11+, pandas 3.1+ verlangen Python >= 3.11
#   * fastapi 0.162+, pydantic 2.14+, aiohttp 3.14+ verlangen Python >= 3.10
#   * aiofiles 25.1+ verlangt Python >= 3.9
# Der Pi (3.12) erfuellt das; aeltere Systeme nicht.
#
# Wichtig fuer den Pi: pandas und matplotlib haben KEINE armv7-Wheels (nur
# x86_64 und aarch64). Auf 32-Bit-Pis bauen sie aus dem Quellcode - dafuer
# sind die Build-Tools noetig (siehe RASPBERRY-PI-DEPLOYMENT.md). Auf dem
# getesteten Pi (aarch64) sind Wheels vorhanden.

# --- Laufzeit ---
aiofiles==25.1.0
aiohttp==3.14.5
# Nicht-blockierender DNS-Resolver (empfohlen). Ohne aiodns nutzt aiohttp den
# ThreadedResolver: jeder Lookup belegt einen Thread des Default-Executors und
# verzoegert damit auch asyncio.to_thread-Aufrufe (Sensor-Lesung), wenn die
# Namensaufloesung stockt (Befund 13.10.2026: Healthcheck-Alarm bei intaktem
# Fernzugriff, dazu Sensor-Timeouts).
#
# aiodns nutzt den System-Resolver aus /etc/resolv.conf. Auf dem Pi verwaltet
# TAILSCALE diese Datei (Nameserver 100.100.100.100 / fd7a:115c:a1e0::53) - ein
# tailscaled-Aussetzer blockiert dann jeden Lookup. Dagegen laesst sich in
# config.ini unter [Netzwerk] DNS_SERVER ein lokal erreichbarer Nameserver
# eintragen (z.B. die Fritz!Box).
#
# Bitte KEINE oeffentlichen Resolver fest verdrahten: sind sie im Netz gesperrt,
# scheitert jeder Lookup mit
# "aiodns.error.DNSError: (12, 'Timeout while contacting DNS servers')".
#
# Edge/armv7 (32-bit): ggf. vorher "sudo apt-get install -y libc-ares-dev".
#
# aiodns 3.6.1 ist die letzte 3.x: aiodns 4.0.0 aendert die API
# (query_dns() statt query(), pycares 5.x) und wird von aiohttp 3.14 noch
# nicht benoetigt. Der Code nutzt ausschliesslich aiohttp.AsyncResolver.
aiodns==3.6.1
fastapi==0.162.2
# api.py nutzt field_validator/model_validator (Pydantic v2).
pydantic==2.14.3
pytz==2026.6
uvicorn==0.54.0
slowapi==0.1.10        # Rate-Limiting fuer schreibende API-Endpoints

# --- Analyse/Diagnose ---
# telegram_charts.py (Chart-Versand); der Import ist bewusst träge (siehe
# Modul-Kommentar: ~70 MB RSS, friert der 512-MB-Pi ein).
matplotlib==3.11.2
# telegram_charts.py (CSV-Auswertung); zieht numpy/dateutil mit.
pandas==3.1.1
